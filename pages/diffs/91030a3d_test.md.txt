+### Test Compose load states
+
+To test how your Compose UI components react to different load states (such as
+`LoadState.Loading` or `LoadState.Error`) in isolation, emit static
+`PagingData` snapshots with explicit [`LoadStates`](https://developer.android.com/reference/kotlin/androidx/paging/LoadStates.md.txt) using
+`PagingData.empty` and `PagingData.from` through a `MutableStateFlow`:
+
+```kotlin
+@get:Rule
+val composeTestRule = createComposeRule()
+
+@Test
+fun userListScreen_displaysLoadingDataAndErrorStates() {
+  val idle = LoadState.NotLoading(endOfPaginationReached = false)
+  // Emit static PagingData snapshots to test UI state transitions without a
+  // Pager or ViewModel.
+  val pagingDataFlow = MutableStateFlow(PagingData.empty<User>())
+
+  composeTestRule.setContent {
+    val lazyPagingItems = pagingDataFlow.collectAsLazyPagingItems()
+    UserListScreen(lazyPagingItems = lazyPagingItems)
+  }
+
+  // Verify the initial loading UI before any items arrive.
+  pagingDataFlow.value = PagingData.empty(
+    sourceLoadStates = LoadStates(
+      refresh = LoadState.Loading,
+      prepend = idle,
+      append = idle
+    )
+  )
+  composeTestRule.onNodeWithTag("LoadingSpinner").assertIsDisplayed()
+
+  // Verify that loaded items replace the loading indicator.
+  pagingDataFlow.value = PagingData.from(
+    data = listOf(User(id = "1", label = "User A")),
+    sourceLoadStates = LoadStates(
+      refresh = idle,
+      prepend = idle,
+      append = idle
+    )
+  )
+  composeTestRule.onNodeWithTag("LoadingSpinner").assertIsNotDisplayed()
+  composeTestRule.onNodeWithText("User A").assertIsDisplayed()
+
+  // Verify that a refresh failure surfaces the error message in the UI.
+  pagingDataFlow.value = PagingData.empty(
+    sourceLoadStates = LoadStates(
+      refresh = LoadState.Error(IOException("Network unavailable")),
+      prepend = idle,
+      append = idle
+    )
+  )
+  composeTestRule.onNodeWithText("Network unavailable").assertIsDisplayed()
+}
+```
